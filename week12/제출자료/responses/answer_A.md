# CSV 정제 함수 `정제(df)`

```python
import pandas as pd


def 정제(df):
    # 원본을 건드리지 않도록 복사본을 만든다
    결과 = df.copy()
    print("시작:", len(결과), "행")

    # 1) 점수가 비어 있는 행을 삭제한다
    결과 = 결과.dropna(subset=["점수"])
    print("1) 빈 점수 삭제 후:", len(결과), "행")

    # 2) 모든 열이 완전히 같은 중복 행은 하나만 남긴다
    결과 = 결과.drop_duplicates()
    print("2) 중복 삭제 후:", len(결과), "행")

    # 3) 반과 과목 값의 앞뒤 공백을 제거한다
    결과["반"] = 결과["반"].str.strip()
    결과["과목"] = 결과["과목"].str.strip()
    print("3) 공백 제거 후:", len(결과), "행")

    # 4) 점수가 0 이상 100 이하인 행만 남긴다
    결과 = 결과[(결과["점수"] >= 0) & (결과["점수"] <= 100)]
    print("4) 범위 밖 점수 삭제 후:", len(결과), "행")

    # 빈 값 때문에 실수형이 된 점수를 정수형으로 되돌린다
    결과["점수"] = 결과["점수"].astype(int)

    # 행 번호를 0부터 다시 매긴다
    결과 = 결과.reset_index(drop=True)
    return 결과


# 사용 예시
df = pd.read_csv("scores_2026.csv", encoding="utf-8", dtype={"학번": str})
깨끗한_df = 정제(df)
print(깨끗한_df.head())
```

## 코드에 대한 설명

- **`dtype={"학번": str}`**: 학번이 `"2026001"`처럼 숫자로만 되어 있으면 pandas가 정수로 읽습니다. 그러면 앞자리의 0이 사라질 수 있습니다. 그래서 문자로 읽도록 지정했습니다.
- **`astype(int)`**: 점수 열에 빈 값(NaN)이 하나라도 있으면 pandas는 그 열 전체를 실수(`85.0`)로 읽습니다. 1단계에서 빈 값을 지운 뒤에만 정수로 바꿀 수 있어서 맨 마지막에 변환했습니다.
- **3단계의 행 수**: 공백 제거는 행을 지우지 않으므로 2단계와 같은 숫자가 출력됩니다. 이것은 정상입니다.

## 확인이 필요한 점

1. **단계 순서와 중복 판정**
   규칙의 순서대로 하면 중복 검사(2단계)를 공백 제거(3단계)보다 먼저 합니다. 이 경우 `'1반'`과 `' 1반'`처럼 공백만 다른 행은 서로 다른 행으로 판단되어 둘 다 남습니다. 공백만 다른 행도 중복으로 보아야 한다면 2단계와 3단계의 순서를 바꾸면 됩니다. 명세에 "완전히 같은 행"이라고 적혀 있어서 코드는 적힌 순서를 그대로 따랐습니다.

2. **반과 과목의 정상 값 검사**
   명세에는 정상 값(`1반`/`2반`, `프로그래밍`/`영어`/`수학`)이 나와 있습니다. 하지만 그 밖의 값을 어떻게 처리할지는 규칙에 없어서 코드에 넣지 않았습니다. 정제가 끝난 뒤 아래 코드로 이상한 값이 남았는지 확인해 보세요.
   ```python
   print(깨끗한_df["반"].unique())
   print(깨끗한_df["과목"].unique())
   ```
